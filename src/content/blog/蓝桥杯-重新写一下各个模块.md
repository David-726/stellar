## 按键处理逻辑
```c
typedef struct key
{
	uint8_t age;
	uint8_t pressed;
	uint8_t long_pressed;
	uint16_t tick;
}key;
key keys[4];
void HAL_Period_...()
{
	if(htim->Instance==TIM6){
	uint16_t sta[4];
	sta[0]=HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_0);
	sta[1]=HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_1);
	sta[2]=HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_2);
	sta[3]=HAL_GPIO_ReadPin(GPIOA,GPIO_PIN_0);
	for (int i=0;i<4;i++)
	{
		switch(keys[i].age)
		{
			case 0:
			if(sta[i]==GPIO_PIN_RESET)
			{
				keys[i].age=1;
			}
			break;
			case 1:
			if(sta[i]==GPIO_PIN_RESET)
			{
				keys[i].tick++;
				if(keys[i].tick>200)
				{
					keys[i].age=0;
				}
			}else
			{
				keys[i].tick=0;
				keys[i].age=0;
			}
			break;
			case 2:
			if(sta[i]==GPIO_PIN_RESET)
			{
				keys[i].tick++;
				if(keys[i].tick>1000)
				{
					keys[i].age=3;
					keys[i].long_pressed=1;
					keys[i].tick=0;
				}else
				{
					keys[i].pressed=1;
					keys[i].tick=0;
					keys[i].pressed=1;
				}
			}
			break;
			case 3:
			if(sta[i]==GPIO_PIN_SET)
			{
				keys[i].age=0;
			}
			break;
		}
	}
	}
}
if(keys[i].pressed){
	keys[i].pressed=0;
}
```

Freq pb4;

{
	if(htim->Instance==TIM2&&htim->Channel==TIM_CHANNEL_1)
	{
		if(cnt==0)

			val1=HAL_TIM_GetCounter(&htim2);

		{
			val2=HAL_TIM_GetCounter(&htim2);
			cnt=0;
		}
			if(val2>val1)val_diff=val2-val1;
			else val_diff=0xFFFFFFFF-val2+val1+1;
			freq.pa15=1000000/val_diff;

	if(htim->Instance==TIM3&&htim->Channel==TIM_CHANNEL_1)

		if(cnt==0)
		{

			val2=HAL_TIM_GetCounter(&htim3);

			if(val2>val1)val_diff=val2-val1;
			else val_diff=0xFFFFFFFF-val2+val1+1;
