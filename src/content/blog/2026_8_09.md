	TIM_InitStruct.TIM_Prescaler = 95;
	TIM_InitStruct.TIM_Period = 65535;
	TIM_InitStruct.TIM_CounterMode = TIM_CounterMode_Up;
	TIM_InitStruct.TIM_ClockDivision = TIM_CKD_DIV1;//何意味
	
	TIM_TimeBaseInit(TIM3,&TIM_InitStruct);
	TIM_Cmd(TIM3,ENABLE);

uint16_t HR04_ReadMm()
{
	uint16_t time_us;
	HR04_Trigger();
	TIM_SetCounter(TIM3,0);
	while(GPIO_ReadInputDataBit(GPIOA,GPIO_Pin_6)==Bit_RESET)
	{
		if(TIM_GetCounter(TIM3)>30000)
		{
			return 0xFFFF;
		}
	}
	TIM_SetCounter(TIM3,0);
	while(GPIO_ReadInputDataBit(GPIOA,GPIO_Pin_6)==Bit_SET)
	{
		if(TIM_GetCounter(TIM3)>30000)
		{
			return 0xFFFF;
		}
	}
	time_us = TIM_GetCounter(TIM3);
	return (uint32_t)time_us*10/58;
}

void Buzzer_SetFrequency(uint32_t frequency)
{
	uint32_t period;
	if(frequency == 0)
	{
		TIM_SetCompare2(TIM2,0);
		return;
	}
	period = 1000000UL /frequency;
	
	TIM_SetAutoload(TIM2,period-1);
	TIM_SetCompare2(TIM2,period/2);
	TIM_GenerateEvent(TIM2,TIM_EventSoure_Update);
}
